El siguiente ejemplo es un archivo `index.html` listo para abrirse en cualquier navegador moderno. Construye un CFDI 4.0 con un concepto gravado, agrega un traslado local con `implocal10`, calcula automáticamente los totales con `SumasConceptosWriter` y permite visualizar o descargar el archivo `.xml`.

```html [index.html]
<!doctype html>
<html lang="es">
  <head>
    <meta charset="UTF-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <title>Generador CFDI 4.0 con IIFE</title>
  </head>
  <body>
    <h1>Generar CFDI 4.0 en el navegador</h1>
    <button id="generar" type="button">Generar y descargar XML</button>
    <pre id="salida"></pre>

    <!-- 1. @nodecfdi/cfdi-core (obligatorio primero) -->
    <script src="https://cdn.jsdelivr.net/npm/@nodecfdi/cfdi-core@^2.0.0/dist/iife/index.iife.js"></script>
    <!-- 2. Utilidades principales de @nodecfdi/cfdi-elements -->
    <script src="https://cdn.jsdelivr.net/npm/@nodecfdi/cfdi-elements/dist/iife/index.iife.js"></script>
    <!-- 3. Elementos CFDI 4.0 e Impuestos Locales 1.0 -->
    <script src="https://cdn.jsdelivr.net/npm/@nodecfdi/cfdi-elements/dist/iife/cfdi40.iife.js"></script>
    <script src="https://cdn.jsdelivr.net/npm/@nodecfdi/cfdi-elements/dist/iife/implocal10.iife.js"></script>

    <script>
      const { nodeToXmlString } = ncCfdiCore;
      const { SumasConceptos, SumasConceptosWriter } = ncIndex;
      const { Comprobante } = ncCfdi40;
      const { ImpuestosLocales } = ncImplocal10;

      document.getElementById('generar').addEventListener('click', () => {
        const comprobante = new Comprobante({
          Serie: 'WEB',
          Folio: '101',
          Fecha: '2026-01-25T12:00:00',
          FormaPago: '03',
          Moneda: 'MXN',
          TipoDeComprobante: 'I',
          Exportacion: '01',
          MetodoPago: 'PUE',
          LugarExpedicion: '64000',
        });

        comprobante.addEmisor({
          Rfc: 'EKU9003173C9',
          Nombre: 'ESCUELA KEMPER URGATE',
          RegimenFiscal: '601',
        });

        comprobante.addReceptor({
          Rfc: 'URE180429TM6',
          Nombre: 'UNIVERSIDAD ROBOTICA ESPAÑOLA',
          DomicilioFiscalReceptor: '65000',
          RegimenFiscalReceptor: '601',
          UsoCFDI: 'G03',
        });

        comprobante
          .addConcepto({
            ClaveProdServ: '90111500',
            Cantidad: '1',
            ClaveUnidad: 'E48',
            Descripcion: 'Servicio de hospedaje',
            ValorUnitario: '1000.00',
            Importe: '1000.00',
            ObjetoImp: '02',
          })
          .addTraslado({
            Base: '1000.00',
            Impuesto: '002',
            TipoFactor: 'Tasa',
            TasaOCuota: '0.160000',
            Importe: '160.00',
          });

        const impLocales = new ImpuestosLocales();
        impLocales.addTrasladosLocales({
          ImpLocTrasladado: 'ISH',
          TasadeTraslado: '3.00',
          Importe: '30.00',
        });
        comprobante.addComplemento(impLocales);

        // Calcular y escribir SubTotal, Total, cfdi:Impuestos e implocal:ImpuestosLocales
        const sumas = new SumasConceptos(comprobante, 2);
        new SumasConceptosWriter(comprobante, sumas, 2).put();

        const xml = nodeToXmlString(comprobante, true);
        document.getElementById('salida').textContent = xml;

        // Descargar el XML generado
        const blob = new Blob([xml], { type: 'application/xml;charset=utf-8' });
        const url = URL.createObjectURL(blob);
        const link = document.createElement('a');
        link.href = url;
        link.download = 'cfdi-40.xml';
        link.click();
        URL.revokeObjectURL(url);
      });
    </script>
  </body>
</html>
```

::tip
Para usar el ejemplo sin conexión a internet, copia los archivos desde `node_modules/@nodecfdi/cfdi-core/dist/iife/` y `node_modules/@nodecfdi/cfdi-elements/dist/iife/` a tu carpeta pública y ajusta las rutas de `<script src="...">`.
::
